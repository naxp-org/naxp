cmake_minimum_required(VERSION 3.23)

project(naxp
	VERSION 0.11.0
	DESCRIPTION "A naxp ('encoded ASCII expression') uses a regex-like syntax to define how ASCII strings should be mapped to a numerical index."
	HOMEPAGE_URL "https://naxp.org"
	LANGUAGES C CXX)

# Tests and the install target are wanted when this is the project being built, and are noise
# when it has been pulled into somebody else's build with add_subdirectory or FetchContent.
option(NAXP_BUILD_TESTS "Build the naxp tests" ${PROJECT_IS_TOP_LEVEL})
option(NAXP_INSTALL "Generate the install target" ${PROJECT_IS_TOP_LEVEL})

# The library is static for now. A shared build on Windows needs export declarations on every
# symbol the C façade exposes, which will come with the façade's first release.
add_library(naxp STATIC
	src/c_emitter.cpp
	src/c_family_emitter.cpp
	src/code_writer.cpp
	src/codec.cpp
	src/compiler.cpp
	src/cpp_emitter.cpp
	src/csharp_emitter.cpp
	src/emitter.cpp
	src/folding.cpp
	src/javascript_emitter.cpp
	src/naxp.cpp
	src/naxp_c.cpp
	src/naxp_message.cpp
	src/padding.cpp
	src/parser.cpp
	src/rank_agreement.cpp
	src/relations.cpp
	src/rx.cpp
	src/rx_converter.cpp
	src/state_map.cpp
	src/tree_walker.cpp
	src/tx.cpp
	src/tx_machine.cpp
	src/value_agreement.cpp
	src/version.cpp
	src/w3_checker.cpp
	src/well_formedness.cpp)

add_library(naxp::naxp ALIAS naxp)

target_compile_features(naxp PUBLIC cxx_std_17)

set_target_properties(naxp PROPERTIES
	CXX_STANDARD 17
	CXX_STANDARD_REQUIRED ON
	CXX_EXTENSIONS OFF)

target_sources(naxp
	PUBLIC
		FILE_SET HEADERS
		BASE_DIRS include
		FILES
			include/naxp/naxp.h
			include/naxp/naxp.hpp)

# The private headers sit beside the sources rather than under include/, so that nothing
# outside the library can reach them.
target_include_directories(naxp PRIVATE src)

target_compile_definitions(naxp PRIVATE NAXP_VERSION="${PROJECT_VERSION}")

include(cmake/warnings.cmake)
naxp_set_warnings(naxp)

# The amalgamation: the library as one naxp.cpp with naxp.h and naxp.hpp beside it, which is
# what the R package vendors and what a release attaches. It is generated by tools/amalgamate.js,
# never edited, and needs Node; where there is none the target is left out, as the fuzz test is.
find_program(NAXP_NODE node)

if(NAXP_NODE)
	set(NAXP_AMALGAMATION_DIR ${CMAKE_CURRENT_BINARY_DIR}/amalgamation)

	set(amalgamation_files
		${NAXP_AMALGAMATION_DIR}/naxp.cpp
		${NAXP_AMALGAMATION_DIR}/naxp.h
		${NAXP_AMALGAMATION_DIR}/naxp.hpp)

	file(GLOB amalgamation_inputs CONFIGURE_DEPENDS src/*.cpp src/*.hpp include/naxp/*)

	add_custom_command(
		OUTPUT ${amalgamation_files}
		COMMAND ${NAXP_NODE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/amalgamate.js --out ${NAXP_AMALGAMATION_DIR}
		DEPENDS tools/amalgamate.js CMakeLists.txt ${amalgamation_inputs}
		COMMENT "Amalgamating naxp ${PROJECT_VERSION}"
		VERBATIM)

	add_custom_target(naxp_amalgamation DEPENDS ${amalgamation_files})
endif()

if(NAXP_BUILD_TESTS)
	include(CTest)
	add_subdirectory(tests)
endif()

if(NAXP_INSTALL)
	include(GNUInstallDirs)
	include(CMakePackageConfigHelpers)

	install(TARGETS naxp
		EXPORT naxpTargets
		FILE_SET HEADERS
		ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
		LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
		RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

	install(EXPORT naxpTargets
		NAMESPACE naxp::
		DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/naxp)

	configure_package_config_file(
		cmake/naxpConfig.cmake.in
		${CMAKE_CURRENT_BINARY_DIR}/naxpConfig.cmake
		INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/naxp)

	write_basic_package_version_file(
		${CMAKE_CURRENT_BINARY_DIR}/naxpConfigVersion.cmake
		COMPATIBILITY SameMajorVersion)

	install(FILES
		${CMAKE_CURRENT_BINARY_DIR}/naxpConfig.cmake
		${CMAKE_CURRENT_BINARY_DIR}/naxpConfigVersion.cmake
		DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/naxp)
endif()
